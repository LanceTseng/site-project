CREATE OR REPLACE FUNCTION public.get_latest_occupied_by_fn(p_eqpt_id integer)
 RETURNS integer
 LANGUAGE plpgsql
 STABLE
AS $function$
DECLARE
    ln_occupied_by INT;
BEGIN
    SELECT h.occupied_by
    INTO ln_occupied_by
    FROM b_eqpt_occupied_his h
    WHERE h.equipment_id = p_eqpt_id
      AND h.released_date IS NULL
    ORDER BY h.occupied_date DESC
    LIMIT 1;

    RETURN ln_occupied_by;
END;
$function$
;

CREATE OR REPLACE FUNCTION public.get_processing_rate_fn(p_parent_task_id integer)
 RETURNS numeric
 LANGUAGE plpgsql
 STABLE
AS $function$
DECLARE
    ln_ct_total INT := 0;
    ln_ct_done INT := 0;
    ln_ct_rate DECIMAL(10,2) := 0;
BEGIN
    -- Get total count of child tasks from v_user_child_task
    SELECT COUNT(1)
    INTO ln_ct_total
    FROM v_user_child_task ct
    WHERE ct.user_parenttask_id = p_parent_task_id;

    IF ln_ct_total > 0 THEN
        -- Get count of completed child tasks from v_user_child_task
        SELECT COUNT(1)
        INTO ln_ct_done
        FROM v_user_child_task ct
        WHERE ct.user_parenttask_id = p_parent_task_id
          AND ct.ct_status_name = 'completed';

        ln_ct_rate := (ln_ct_done::DECIMAL / ln_ct_total) * 100;

    ELSE
        -- Fallback to rel_user_parenttask table if no child tasks found
        SELECT COUNT(1)
        INTO ln_ct_total
        FROM rel_user_parenttask ct
        WHERE ct.id = p_parent_task_id;

        SELECT COUNT(1)
        INTO ln_ct_done
        FROM rel_user_parenttask ct
        WHERE ct.id = p_parent_task_id
          AND ct.status = 2;

        IF ln_ct_total > 0 THEN
            ln_ct_rate := (ln_ct_done::DECIMAL / ln_ct_total) * 100;
        ELSE
            ln_ct_rate := 0;
        END IF;
    END IF;

    RETURN ROUND(ln_ct_rate, 2);
END;
$function$
;
CREATE OR REPLACE FUNCTION public.get_task_group_id_fn(p_task_id integer)
 RETURNS integer
 LANGUAGE plpgsql
 STABLE
AS $function$
DECLARE
    p_group_id INT := -1;
BEGIN
    SELECT COALESCE(task_group_id, -1)
    INTO p_group_id
    FROM b_parent_tasks
    WHERE task_id = p_task_id
    LIMIT 1;

    RETURN p_group_id;
END;

CREATE OR REPLACE PROCEDURE public.remove_task_by_user(IN p_user integer, IN p_group integer)
 LANGUAGE plpgsql
AS $procedure$DECLARE
    child_count INT := 0;
    parent_count INT := 0;
BEGIN
    -- Delete child tasks
    DELETE FROM rel_user_childtask c
    WHERE EXISTS (
        SELECT 1
        FROM v_user_task v
        WHERE v.pt_group_id = p_group
          AND v.user_id = p_user
          AND v.line_id = c.id
    );
    GET DIAGNOSTICS child_count = ROW_COUNT;

    -- Delete parent tasks
    DELETE FROM rel_user_parenttask p
    WHERE EXISTS (
        SELECT 1
        FROM v_user_task v
        WHERE v.pt_group_id = p_group
          AND v.user_id = p_user
		  and v.head_id = p.id
    );
    GET DIAGNOSTICS parent_count = ROW_COUNT;

    RAISE NOTICE 'Deleted % child rows, % parent rows', child_count, parent_count;
END;
$procedure$
;
